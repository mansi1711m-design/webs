#include <stdio.h>

void main()
{

int a[100], n, i;

printf("Enter the size of array: ");
scanf("%d", &n);
printf("Enter %d elements:\n", n);
for(i = 0; i < n; i++)
{
scanf("%d", &a[i]);
}
printf("Array in reverse order:\n");
for(i = n - 1; i >= 0; i--)
{
printf("%d ", a[i]);
}
}

Output

Enter the size of array: 5
Enter 5 elements:
10 20 30 40 50
Array in reverse order:
50 40 30 20 10
