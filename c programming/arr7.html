#include <stdio.h>
void main()
{
int n, i;
printf("Enter the size of array: ");
scanf("%d", &n);
int arr[n];
printf("Enter %d elements:\n", n);
for(i = 0; i < n; i++)
{
scanf("%d", &arr[i]);
}
for(i = 0; i < n; i++)
{
arr[i] = arr[i] * arr[i];
}
printf("Array after squaring:\n");
for(i = 0; i < n; i++)
{
printf("%d ", arr[i]);
}
}

Output

Enter the size of array: 5
Enter 5 elements:
2 3 4 5 6
Array after squaring:
4 9 16 25 36
