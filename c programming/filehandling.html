Q2. Enter a file name from user and the data from user and put it in file.but the data should be in capital.
Answer:

#include <stdio.h>
int main()
{
FILE *fp;
char filename[50];
char data[200];
int i;
printf("Enter file name: ");
scanf("%s", filename);
fp = fopen(filename, "w");
if (fp == NULL)
{
printf("File cannot be created!");
return 1;
}
printf("Enter data: ");
getchar();
fgets(data, sizeof(data), stdin);
for (i = 0; data[i] != '\0'; i++)
{
data[i] = toupper(data[i]);
}
fprintf(fp, "%s", data);
fclose(fp);
printf("Data successfully stored in capital letters.");
return 0;
}

Output

Enter file name: student.txt
Enter data: hello welcome to c programming

Q7. Create a file count no. of characters in a file.
Answer:

void main()
{FILE *fp;
char fname[30];
char ch;
int count=0;
clrscr();
printf("Enter file name: ");
scanf("%s",fname);
fp=fopen(fname,"w");
printf("Enter data: ");
fflush(stdin);
while((ch=getchar())!='\n')
{fputc(ch,fp);
}
fclose(fp);
fp=fopen(fname,"r");
while((ch=fgetc(fp))!=EOF)
{ count++;
}
fclose(fp);
printf("\nTotal characters = %d",count);
}
