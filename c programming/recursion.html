Q2 Write a program of recursion with global variable and display the output.
Answer:

#include<stdio.h>
#include<conio.h>
#include<dos.h>
{
int num=1;
void display()
{
if(num<=5)
{
printf("%d\n",num);
num++;
display();
}
printf("A");
}
void main()
{
clrscr();
display();
}
Output:
1
2
3
4
5
AAAAAA
